import os, httpx
from dotenv import load_dotenv
load_dotenv()
key = os.getenv("GMS_API_KEY") or os.getenv("OPENAI_API_KEY") or ""
assert key, "FAIL: .env에서 GMS 키 못 찾음 (키 이름 확인 필요)"
r = httpx.post(
    "https://gms.ssafy.io/gmsapi/api.openai.com/v1/embeddings",
    headers={"Authorization": f"Bearer {key}"},
    json={"model": "text-embedding-3-small", "input": "키 생존 테스트"},
    timeout=15,
)
print(f"HTTP {r.status_code}")
print("RESULT: PASS - 현재 키 유효" if r.status_code == 200
      else f"RESULT: FAIL - 키 죽음/거부: {r.text[:200]}")
PY
